\documentclass[border=10pt]{standalone}
\usepackage[UTF8]{ctex}
\usepackage{tikz}
\usetikzlibrary{arrows.meta,positioning,calc,decorations.pathreplacing}
\usepackage{amsmath}

\begin{document}
\begin{tikzpicture}[
    node distance=1.8cm and 2.5cm,
    block/.style={rectangle, draw=#1, fill=#1!8, rounded corners=3pt,
        minimum width=2.8cm, minimum height=1.2cm, align=center,
        font=\small\bfseries, line width=1pt},
    arrow/.style={-{Stealth[length=5pt]}, line width=1pt, #1},
    lbl/.style={font=\scriptsize, #1},
]

\node[block=teal] (soc_t) {$SOC(t)$};
\node[below=0.1cm of soc_t, font=\tiny\color{gray}] {储能荷电状态};

\node[block=teal, right=4cm of soc_t] (soc_t1) {$SOC(t{+}1)$};
\node[below=0.1cm of soc_t1, font=\tiny\color{gray}] {下一时段状态};

\node[block=blue!70!black, above=1.8cm of $(soc_t)!0.5!(soc_t1)$] (charge) {充电过程};
\node[below=0.05cm of charge, font=\tiny\color{blue!60!black}] {$P_{\mathrm{ch}}(t) \cdot \eta_{\mathrm{ch}}$};

\node[block=red!70!black, below=2.2cm of $(soc_t)!0.5!(soc_t1)$] (discharge) {放电过程};
\node[above=0.05cm of discharge, font=\tiny\color{red!60!black}] {$P_{\mathrm{dis}}(t) / \eta_{\mathrm{dis}}$};

\node[block=orange!80!black, left=1.5cm of charge] (renewable) {风光余电};
\node[below=0.05cm of renewable, font=\tiny\color{orange!60!black}] {$P_{\mathrm{RE}} - P_{\mathrm{load}}$};

\node[block=purple!70!black, left=1.5cm of discharge] (load) {补充负荷};
\node[above=0.05cm of load, font=\tiny\color{purple!60!black}] {$P_{\mathrm{load}} - P_{\mathrm{RE}}$};

% Main state transition arrow
\draw[arrow=teal!70!black] (soc_t) -- node[above, lbl=black]{状态转移} (soc_t1);

% Charge path
\draw[arrow=orange!80!black] (renewable) -- (charge);
\draw[arrow=blue!70!black] (charge) -| (soc_t1);

% Discharge path
\draw[arrow=red!70!black] (soc_t) |- (discharge);
\draw[arrow=purple!70!black] (discharge) -- (load);

% Constraints box
\node[right=1.5cm of soc_t1, draw=gray!50, rounded corners=2pt,
      fill=gray!5, align=left, font=\scriptsize, inner sep=6pt] (cbox) {
    $SOC_{\min} \leq SOC(t) \leq SOC_{\max}$\\[2pt]
    $0 \leq P_{\mathrm{ch}}(t) \leq P_{\mathrm{ch,max}}$\\[2pt]
    $0 \leq P_{\mathrm{dis}}(t) \leq P_{\mathrm{dis,max}}$\\[2pt]
    $SOC(0) = SOC(24) = SOC_{\mathrm{init}}$
};
\node[above=0.1cm of cbox, font=\small\bfseries\color{gray!70!black}] {约束条件};

% State equation at bottom
\node[below=3.5cm of $(soc_t)!0.5!(soc_t1)$, draw=teal!30, rounded corners=2pt,
      fill=teal!3, inner sep=8pt, font=\small] (eq) {
    $SOC(t{+}1) = SOC(t) + \frac{P_{\mathrm{ch}}(t) \cdot \eta_{\mathrm{ch}} \cdot \Delta t}{E_{\mathrm{cap}}} - \frac{P_{\mathrm{dis}}(t) \cdot \Delta t}{\eta_{\mathrm{dis}} \cdot E_{\mathrm{cap}}}$
};
\node[above=0.1cm of eq, font=\small\bfseries\color{teal!70!black}] {储能状态转移方程};

\end{tikzpicture}
\end{document}
