import pandas as pd
import numpy as np

def load_load_curve(filepath):
    """加载常规电负荷标幺功率曲线"""
    df = pd.read_excel(filepath)
    load_pu = df.iloc[:, 1].values  # 24h标幺值
    return load_pu * 6.0  # MW, 峰值6MW

def load_wind_pv_typical(filepath):
    """加载典型日风光标幺功率曲线"""
    df = pd.read_excel(filepath)
    wind_pu = df.iloc[:, 1].values
    pv_pu = df.iloc[:, 2].values
    return wind_pu * 40.0, pv_pu * 64.0  # MW

def load_scenarios(wind_file, pv_file):
    """加载6种风电*4种光伏场景"""
    df_w = pd.read_excel(wind_file)
    df_p = pd.read_excel(pv_file)
    scenarios = []
    for wi in range(6):
        for pi in range(4):
            wind = df_w.iloc[:, wi+1].values * 40.0
            pv = df_p.iloc[:, pi+1].values * 64.0
            scenarios.append((wind, pv))
    return scenarios

def load_price():
    """分时电价映射"""
    price = np.zeros(24)
    valley = [23, 0, 1, 2, 3, 4, 5, 6]  # 0.3424
    peak = [10, 11, 12, 13, 14, 18, 19, 20]  # 0.8024
    for t in range(24):
        if t in valley:
            price[t] = 0.3424
        elif t in peak:
            price[t] = 0.8024
        else:
            price[t] = 0.6074
    return price

\begin{lstlisting}[language=Python, basicstyle=\ttfamily\small, breaklines=true]
from pulp import *

def solve_q2_milp(wind, pv, load, Q_target, price, sell_price=0.3779):
    P_H2NH3 = 41.5  # MW
    r_NH3 = 3.0     # t/h
    k = int(Q_target / r_NH3)
    C_OM_h = 5003   # yuan/h
    
    prob = LpProblem("Q2_Discrete", LpMinimize)
    u = [LpVariable(f"u_{t}", cat='Binary') for t in range(24)]
    P_buy = [LpVariable(f"Pb_{t}", lowBound=0) for t in range(24)]
    
    # 产量约束
    prob += lpSum(u) == k
    
    # 功率平衡
    for t in range(24):
        P_net = load[t] + u[t] * P_H2NH3 - wind[t] - pv[t]
        prob += P_buy[t] >= P_net
    
    # 目标函数
    C_buy = lpSum(price[t] * P_buy[t] * 1000 for t in range(24))
    C_OM = lpSum(u[t] * C_OM_h for t in range(24))
    P_sell = [P_buy[t] - (load[t] + u[t]*P_H2NH3 - wind[t] - pv[t])
              for t in range(24)]
    I_sell = sell_price * lpSum(P_sell[t] * 1000 for t in range(24))
    
    prob += C_buy + C_OM - I_sell
    prob.solve(PULP_CBC_CMD(msg=0))
    
    return {
        'schedule': [int(value(u[t])) for t in range(24)],
        'cost': value(prob.objective),
        'C_ton': value(prob.objective) / Q_target
    }
\end{lstlisting}

def solve_q3_lp(wind, pv, load, Q_target, price, sell_price=0.3779):
    P_H2NH3 = 41.5
    r_NH3 = 3.0
    alpha_sum = Q_target / r_NH3
    C_OM_h = 5003
    
    prob = LpProblem("Q3_Continuous", LpMinimize)
    alpha = [LpVariable(f"a_{t}", 0.1, 1.0) for t in range(24)]
    P_buy = [LpVariable(f"Pb_{t}", lowBound=0) for t in range(24)]
    
    # 产量约束
    prob += lpSum(alpha) == alpha_sum
    
    # 功率平衡
    for t in range(24):
        P_net = load[t] + alpha[t] * P_H2NH3 - wind[t] - pv[t]
        prob += P_buy[t] >= P_net
    
    # 目标函数
    C_buy = lpSum(price[t] * P_buy[t] * 1000 for t in range(24))
    C_OM = lpSum(alpha[t] * C_OM_h for t in range(24))
    P_sell = [P_buy[t] - (load[t] + alpha[t]*P_H2NH3 - wind[t] - pv[t])
              for t in range(24)]
    I_sell = sell_price * lpSum(P_sell[t] * 1000 for t in range(24))
    
    prob += C_buy + C_OM - I_sell
    prob.solve(PULP_CBC_CMD(msg=0))
    
    return {
        'alpha': [value(alpha[t]) for t in range(24)],
        'C_ton': value(prob.objective) / Q_target
    }

\subsection{问题四储能优化}
\begin{lstlisting}[language=Python, basicstyle=\ttfamily\small, breaklines=true]
def solve_q4_storage(wind, pv, load, C_sto, eta_c=0.9, eta_d=0.9,
                     sigma=0.002):
    P_H2NH3 = 41.5
    P_sto_max = C_sto / 2.0  # 0.5C
    
    prob = LpProblem("Q4_Storage", LpMaximize)
    alpha = [LpVariable(f"a_{t}", 0, 1.0) for t in range(24)]
    P_ch = [LpVariable(f"Pc_{t}", 0, P_sto_max) for t in range(24)]
    P_dis = [LpVariable(f"Pd_{t}", 0, P_sto_max) for t in range(24)]
    E_sto = [LpVariable(f"E_{t}", 0, C_sto) for t in range(25)]
    
    # 功率平衡（离网）
    for t in range(24):
        prob += (alpha[t] * P_H2NH3 + load[t] + P_ch[t] 
                 <= wind[t] + pv[t] + P_dis[t])
        prob += alpha[t] >= 0.1  # 最低负荷率，可通过v(t)松弛
    
    # 储能SOC动态
    for t in range(24):
        prob += (E_sto[t+1] == E_sto[t] * (1-sigma) 
                 + eta_c * P_ch[t] - P_dis[t] / eta_d)
    
    # 日循环约束
    prob += E_sto[24] == E_sto[0]
    
    # 最大化产量
    prob += lpSum(alpha[t] for t in range(24)) * 3.0
    prob.solve(PULP_CBC_CMD(msg=0))
    
    Q_NH3 = sum(value(alpha[t]) for t in range(24)) * 3.0
    return {'Q_NH3': Q_NH3, 'C_sto': C_sto}
\end{lstlisting}
